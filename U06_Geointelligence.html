Zuordnung der Filme zu Regalen

Führen Sie folgendes Statement auf Ihrer Instanz des SQL Server aus:

						
							ALTER TABLE vrs.inventory ADD shelve_id INT NULL;

							ALTER TABLE vrs.inventory
								ADD CONSTRAINT FK_Inventory_Shelve FOREIGN KEY (shelve_id)
									REFERENCES vrs.shelve (shelve_id);

							UPDATE i
							SET i.shelve_id = t.shelve_id 
							FROM vrs.inventory i
							JOIN (
								SELECT	t2.inventory_id,
										NTILE(12) OVER(ORDER BY t2.film_id DESC) AS shelve_id
								FROM vrs.inventory t2
								WHERE t2.store_id = 1
							) t
								ON t.inventory_id = i.inventory_id;

Laden von Bewegungsdaten

Führen Sie folgendes Statement auf Ihrer Instanz des SQL Server aus:

								
									SELECT movement.*
									INTO dbo.movements
									FROM OPENROWSET (
											BULK 'csvcontainer/google_takeout.json',
											DATA_SOURCE = 'azureblobstorageDS',
											SINGLE_CLOB
										) AS j
									CROSS APPLY OPENJSON(BulkColumn, '$.locations')
									WITH (
											timestampMs BIGINT,
											latitudeE7 BIGINT,
											longitudeE7 BIGINT,
											accuracy int
										) AS movement;
		
									ALTER TABLE dbo.movements
										ALTER COLUMN timestampMs BIGINT NOT NULL;
		
									ALTER TABLE dbo.movements
										ADD recording_time DATETIME2 NULL,
											latitude DECIMAL(10,7) NULL,
											longitude DECIMAL(10,7) NULL,
											location_point GEOGRAPHY NULL,
											CONSTRAINT PK_movements PRIMARY KEY CLUSTERED (timestampMs);
		
									UPDATE dbo.movements
										SET recording_time = DATEADD(SECOND, timestampMs / 1000, '1970-1-1'),
											latitude = CAST(CAST(latitudeE7 AS DECIMAL(10,0)) / POWER(10,7) AS decimal(10,7)),
											longitude = CAST(CAST(longitudeE7 AS DECIMAL(10,0)) / POWER(10,7) AS decimal(10,7));
		
									UPDATE dbo.movements
										SET location_point = geography::Point(latitude, longitude , 4326);

Aufgabe: Auswertung der Umsätze pro Regal

Führen Sie folgendes Statement auf Ihrer Instanz des SQL Server per SSMS aus:

							
								WITH cte_payment_amount AS
								(
									SELECT i.shelve_id,
										SUM(p.amount) AS sum_amount
									FROM [vrs].[payment] p
									JOIN [vrs].[rental] r
										ON r.rental_id = p.rental_id
									JOIN [vrs].[inventory] i
										ON i.inventory_id = r.inventory_id
									GROUP BY i.shelve_id
								)
								SELECT cte.shelve_id,
										cte.sum_amount,
										s.shelve_geo
								FROM cte_payment_amount cte
								JOIN vrs.shelve s
									ON s.shelve_id = cte.shelve_id
