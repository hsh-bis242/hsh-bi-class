Zuordnung der Filme zu Regalen

Führen Sie folgendes Statement auf Ihrer Instanz des SQL Server aus:

						
							ALTER TABLE vrs.inventory ADD shelve_id INT NULL;

							ALTER TABLE vrs.inventory
								ADD CONSTRAINT FK_Inventory_Shelve FOREIGN KEY (shelve_id)
									REFERENCES vrs.shelve (shelve_id);

							UPDATE i
							SET i.shelve_id = t.shelve_id 
							FROM vrs.inventory i
							JOIN (
								SELECT	t2.inventory_id,
										NTILE(12) OVER(ORDER BY t2.film_id DESC) AS shelve_id
								FROM vrs.inventory t2
								WHERE t2.store_id = 1
							) t
								ON t.inventory_id = i.inventory_id;

Laden von Bewegungsdaten

Führen Sie folgendes Statement auf Ihrer Instanz des SQL Server aus:

								
									SELECT movement.*
									INTO dbo.movements
									FROM OPENROWSET (
											BULK 'csvcontainer/google_takeout.json',
											DATA_SOURCE = 'azureblobstorageDS',
											SINGLE_CLOB
										) AS j
									CROSS APPLY OPENJSON(BulkColumn, '$.locations')
									WITH (
											timestampMs BIGINT,
											latitudeE7 BIGINT,
											longitudeE7 BIGINT,
											accuracy int
										) AS movement;
		
									ALTER TABLE dbo.movements
										ALTER COLUMN timestampMs BIGINT NOT NULL;
		
									ALTER TABLE dbo.movements
										ADD recording_time DATETIME2 NULL,
											latitude DECIMAL(10,7) NULL,
											longitude DECIMAL(10,7) NULL,
											location_point GEOGRAPHY NULL,
											CONSTRAINT PK_movements PRIMARY KEY CLUSTERED (timestampMs);
		
									UPDATE dbo.movements
										SET recording_time = DATEADD(SECOND, timestampMs / 1000, '1970-1-1'),
											latitude = CAST(CAST(latitudeE7 AS DECIMAL(10,0)) / POWER(10,7) AS decimal(10,7)),
											longitude = CAST(CAST(longitudeE7 AS DECIMAL(10,0)) / POWER(10,7) AS decimal(10,7));
		
									UPDATE dbo.movements
										SET location_point = geography::Point(latitude, longitude , 4326);

Aufgabe: Auswertung der Umsätze pro Regal

Führen Sie folgendes Statement auf Ihrer Instanz des SQL Server per SSMS aus:

							
								WITH cte_payment_amount AS
								(
									SELECT i.shelve_id,
										SUM(p.amount) AS sum_amount
									FROM [vrs].[payment] p
									JOIN [vrs].[rental] r
										ON r.rental_id = p.rental_id
									JOIN [vrs].[inventory] i
										ON i.inventory_id = r.inventory_id
									GROUP BY i.shelve_id
								)
								SELECT cte.shelve_id,
										cte.sum_amount,
										s.shelve_geo
								FROM cte_payment_amount cte
								JOIN vrs.shelve s
									ON s.shelve_id = cte.shelve_id

Aufgabe: Auswertung von Bewegungsdaten

Ein großer Teil des Datensatzes besteht aus Stillstandsdaten. Allein 71500 Datensätze liegen am Wohnort:

								
									SELECT *
									FROM [dbo].[movements] m
									WHERE
									m.location_point.STIntersects(
										geography::Point(52.3561617, 9.7539098 , 4326)
										.STBuffer(15)) = 1;

Deshalb sollte man zunächst "echte" Ausflüge / Trips identifizieren.

Aufgabe: Auswertung von Bewegungsdaten

Zusammenfassen der Daten zu Trips

						
							WITH cte_diffs AS
							(
								SELECT	m.timestampMs,
										m.recording_time,
										m.location_point,
										DATEDIFF(
											SECOND,
											LAG(m.recording_time) OVER (ORDER BY m.timestampMs),
											m.recording_time
										) AS recording_time_diff_second,
										m.location_point.STDistance(LAG(m.location_point,1) OVER (ORDER BY m.timestampMs ASC)) AS distance_to_predecessor
								FROM movements m
								WHERE m.latitudeE7 <> 0
								AND m.longitudeE7 <> 0
								AND m.accuracy < 500
							),
							cte_trips AS
							(
								SELECT	cte.timestampMs,
										cte.recording_time,
										cte.location_point,
										cte.recording_time_diff_second,
										cte.distance_to_predecessor,
										SUM
										(
											CASE
											-- Ein neuer Trip wird definiert, wenn zwischen den Datenpaketen entweder 30min liegen...
												WHEN cte.recording_time_diff_second > 1800
												OR 
												(
													-- Oder in 15min weniger als 3000m zurückgelegt werden.
													cte.recording_time_diff_second > 900
													AND cte.distance_to_predecessor < 3000
												) THEN 1
												ELSE 0
											END
										) OVER (ORDER BY cte.timestampMs ROWS UNBOUNDED PRECEDING) AS trip
								FROM cte_diffs cte
								WHERE cte.distance_to_predecessor > 0
								AND cte.recording_time_diff_second > 0
								AND cte.distance_to_predecessor / cte.recording_time_diff_second > 0.8
							),
							cte_velocity AS 
							(
								SELECT	cte.timestampMs,
										cte.recording_time,
										cte.location_point,
										cte.recording_time_diff_second,
										cte.distance_to_predecessor,
										cte.trip
								FROM	cte_trips cte
								WHERE cte.recording_time_diff_second > 0
							)
							SELECT *
							FROM 
							(
								SELECT	DATEADD(SECOND, SUM(cte.recording_time_diff_second)*-1, MAX(cte.recording_time)) as recording_time_start,
										MAX(cte.recording_time) as recording_time_end,
										geography::CollectionAggregate(cte.location_point) as geo,
										SUM(cte.recording_time_diff_second) as diff_Seconds,
										SUM(cte.distance_to_predecessor) as distance,
										cte.trip,
										(((SUM(cte.distance_to_predecessor))/(SUM(cte.recording_time_diff_second)))*3.6) AS kmh,
										count(*) as items_count
								FROM cte_velocity cte
								GROUP BY trip) s
							WHERE kmh < 300;
